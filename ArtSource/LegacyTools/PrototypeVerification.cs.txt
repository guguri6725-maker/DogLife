using System.Collections;
using UnityEngine;

namespace DogLife
{
    // 저장된 씬을 바꾸지 않고 이동과 무기 기능을 플레이 모드에서 검사합니다.
    public sealed class PrototypeVerification : MonoBehaviour
    {
        public string Result { get; private set; } = "RUNNING";
        public int Passed { get; private set; }
        // 조건이 실패하면 즉시 예외를 발생시켜 실패 원인을 남깁니다.
        void Check(bool condition, string message)
        {
            if (!condition) { Result = "FAIL: " + message; throw new System.Exception(Result); }
            Passed++; Debug.Log("VERIFY PASS: " + message);
        }
        IEnumerator Start()
        {
            var motor = FindFirstObjectByType<PlayerMotor>(); var weapons = motor.GetComponent<PlayerWeapons>();
            motor.readInput = false; weapons.readInput = false;
            foreach (var practice in FindObjectsByType<CombatHealth>(FindObjectsSortMode.None)) if (practice.practiceTarget) practice.gameObject.SetActive(false);
            yield return new WaitForSeconds(.6f);
            Check(motor.Grounded, "Grounded on 3D floor");
            float x = motor.transform.position.x;
            motor.MoveInput = -1; yield return new WaitForSeconds(.4f); motor.MoveInput = 0;
            Check(motor.transform.position.x < x - 1, "Horizontal movement");
            Check(Mathf.Abs(motor.transform.position.z) < .001f, "Locked side-view depth");
            float y = motor.transform.position.y; motor.RequestJump(); yield return new WaitForSeconds(.25f);
            Check(motor.transform.position.y > y + .8f, "Jump height");
            yield return new WaitForSeconds(1.6f); Check(motor.Grounded, "Landing");
            Check(motor.RequestDash(), "Dash starts"); Check(!motor.RequestDash(), "Dash cooldown blocks repetition");
            yield return new WaitForSeconds(.3f); Check(!motor.IsDashing, "Dash ends");
            var target = GameObject.CreatePrimitive(PrimitiveType.Capsule); target.name = "Verification target";
            target.transform.position = motor.transform.position + new Vector3(3, 0, 0);
            var hp = target.AddComponent<CombatHealth>(); hp.maxHealth = 100; hp.Restore();
            weapons.Equip(0); weapons.AimAt(target.transform.position); Check(weapons.TryAttack(), "Rifle fires");
            Check(!weapons.TryAttack(), "Rifle cadence enforced");
            yield return new WaitForSeconds(.2f); Check(hp.Current == 97, "Rifle projectile damages target once");
            weapons.Equip(1); weapons.AimAt(target.transform.position); weapons.TryAttack();
            Check(FindObjectsByType<CombatProjectile>(FindObjectsSortMode.None).Length == 5, "Shotgun emits five pellets");
            yield return new WaitForSeconds(.3f); Check(hp.Current < 97, "Shotgun hit");
            foreach (var p in FindObjectsByType<CombatProjectile>(FindObjectsSortMode.None)) Destroy(p.gameObject);
            target.transform.position = motor.transform.position + new Vector3(1.2f, 0, 0); Physics.SyncTransforms();
            weapons.Equip(2); weapons.AimAt(target.transform.position); float before = hp.Current; weapons.TryAttack();
            Check(hp.Current == before - 3, "Sword melee damages once");
            target.transform.position = motor.transform.position + new Vector3(3, 0, 0); Physics.SyncTransforms();
            weapons.Equip(3); weapons.AimAt(target.transform.position); before = hp.Current; weapons.TryAttack(1);
            yield return new WaitForSeconds(.4f); Check(hp.Current == before - 20, "Charged sword wave damages target");
            var wall = GameObject.CreatePrimitive(PrimitiveType.Cube); wall.transform.position = motor.transform.position + Vector3.right * 1.5f; wall.transform.localScale = new Vector3(.1f, 4, 4); wall.layer = 8; Physics.SyncTransforms();
            weapons.Equip(0); before = hp.Current; weapons.TryAttack(); yield return new WaitForSeconds(.3f);
            Check(hp.Current == before, "Thin wall stops high-speed projectiles");
            Destroy(wall); Destroy(target); motor.Respawn(); motor.readInput = true; weapons.readInput = true;
            Result = "PASS: " + Passed + " checks"; Debug.Log(Result);
        }
    }
}
