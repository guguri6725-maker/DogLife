using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;

namespace DogLife.Editor
{
    // 프로토타입 씬에서 공통으로 사용하는 재질, 지형과 플레이어를 생성합니다.
    public static class PrototypeBuilder
    {
        public const string Root = "Assets/DogLife";
        public static Material MakeMaterial(string name, Color color)
        {
            string path = Root + "/Materials/" + name + ".mat";
            var mat = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (!mat) { mat = new Material(Shader.Find("Standard")); AssetDatabase.CreateAsset(mat, path); }
            mat.color = color; mat.SetFloat("_Glossiness", .18f); EditorUtility.SetDirty(mat); return mat;
        }
        public static GameObject Shape(string name, PrimitiveType type, Vector3 position, Vector3 scale, Material material, Transform parent = null, bool solid = true)
        {
            var go = GameObject.CreatePrimitive(type); go.name = name;
            go.transform.SetParent(parent, false); go.transform.localPosition = position; go.transform.localScale = scale;
            go.GetComponent<Renderer>().sharedMaterial = material;
            if (!solid) Object.DestroyImmediate(go.GetComponent<Collider>());
            else go.layer = 8;
            return go;
        }
        public static void Folders()
        {
            foreach (var name in new[] { "Materials", "Prefabs", "Data", "Scenes" })
                if (!AssetDatabase.IsValidFolder(Root + "/" + name)) AssetDatabase.CreateFolder(Root, name);
        }
        // 테스트에 필요한 기본 플레이어와 무기 구성을 만듭니다.
        public static GameObject CreatePlayer()
        {
            var player = new GameObject("Player"); player.tag = "Player"; player.transform.position = new Vector3(0, 1.1f, 0);
            var capsule = player.AddComponent<CapsuleCollider>(); capsule.height = 2; capsule.radius = .38f;
            var friction = new PhysicsMaterial("Player friction") { dynamicFriction = 0, staticFriction = 0, frictionCombine = PhysicsMaterialCombine.Minimum };
            string physicsPath = Root + "/Materials/PlayerFriction.physicMaterial";
            var existing = AssetDatabase.LoadAssetAtPath<PhysicsMaterial>(physicsPath);
            if (existing) { Object.DestroyImmediate(friction); friction = existing; } else AssetDatabase.CreateAsset(friction, physicsPath);
            capsule.material = friction;
            player.AddComponent<Rigidbody>();
            var motor = player.AddComponent<PlayerMotor>();
            var visuals = new GameObject("VisualRoot_REPLACE_MODEL").transform; visuals.SetParent(player.transform, false); motor.visualRoot = visuals;
            var coat = MakeMaterial("Player teal", new Color(.12f, .65f, .65f));
            var dark = MakeMaterial("Dark", new Color(.07f, .1f, .14f));
            Shape("Temporary body", PrimitiveType.Capsule, Vector3.zero, new Vector3(.72f, .75f, .65f), coat, visuals, false);
            Shape("Temporary head", PrimitiveType.Sphere, new Vector3(0, .67f, 0), Vector3.one * .65f, coat, visuals, false);
            Shape("Muzzle", PrimitiveType.Cube, new Vector3(.3f, .64f, -.03f), new Vector3(.42f, .22f, .38f), dark, visuals, false);
            Shape("Ear", PrimitiveType.Cube, new Vector3(-.12f, 1, 0), new Vector3(.2f, .4f, .25f), coat, visuals, false);
            var socket = new GameObject("WeaponSocket").transform; socket.SetParent(player.transform, false); socket.localPosition = new Vector3(0, .25f, 0);
            var weapons = player.AddComponent<PlayerWeapons>(); weapons.weaponSocket = socket;
            weapons.projectileMaterial = MakeMaterial("Projectile", Color.yellow);
            weapons.loadout = new WeaponDefinition[4];
            string[] names = { "Rifle", "Shotgun", "Sword", "ChargeSword" };
            Color[] colors = { new Color(1, .72f, .2f), new Color(.95f, .4f, .16f), new Color(.65f, .85f, 1), new Color(.75f, .4f, 1) };
            for (int i = 0; i < 4; i++)
            {
                string path = Root + "/Data/" + names[i] + ".asset";
                var data = AssetDatabase.LoadAssetAtPath<WeaponDefinition>(path);
                if (!data) { data = ScriptableObject.CreateInstance<WeaponDefinition>(); AssetDatabase.CreateAsset(data, path); }
                data.displayName = names[i]; data.kind = (WeaponKind)i; data.color = colors[i];
                data.damage = new[] { 3f, 4f, 3f, 20f }[i]; data.cooldown = new[] { .2f, 2f, .3f, 1.8f }[i];
                data.projectileSpeed = new[] { 30f, 20f, 0f, 14f }[i]; data.pellets = i == 1 ? 5 : 1; data.spread = i == 1 ? 60 : 0;
                data.range = i == 1 ? 18 : 40;
                var model = new GameObject(names[i] + "_REPLACE_MODEL");
                var mat = MakeMaterial(names[i], colors[i]);
                Shape("Grip", PrimitiveType.Cube, new Vector3(.25f, -.15f, 0), new Vector3(.18f, .35f, .2f), dark, model.transform, false);
                Shape(i < 2 ? "Barrel" : "Blade", PrimitiveType.Cube, new Vector3(i < 2 ? .6f : .9f, 0, 0), new Vector3(i < 2 ? 1 : 1.6f, i < 2 ? .2f : .14f, .17f), mat, model.transform, false);
                data.visualPrefab = PrefabUtility.SaveAsPrefabAsset(model, Root + "/Prefabs/" + names[i] + ".prefab");
                Object.DestroyImmediate(model); weapons.loadout[i] = data; EditorUtility.SetDirty(data);
            }
            var health = player.AddComponent<CombatHealth>(); health.player = true; health.maxHealth = 100;
            PrefabUtility.SaveAsPrefabAsset(player, Root + "/Prefabs/Player.prefab");
            return player;
        }
        public static PrototypeHUD Environment(GameObject player, float cameraMax)
        {
            var cameraObject = new GameObject("Main Camera"); cameraObject.tag = "MainCamera";
            var camera = cameraObject.AddComponent<Camera>(); camera.orthographic = true; camera.orthographicSize = 6.5f;
            camera.clearFlags = CameraClearFlags.SolidColor; camera.backgroundColor = new Color(.055f, .085f, .13f);
            cameraObject.transform.position = new Vector3(0, 4, -18); cameraObject.transform.rotation = Quaternion.Euler(8, 0, 0);
            cameraObject.AddComponent<AudioListener>();
            var follow = cameraObject.AddComponent<PlayerCamera>(); follow.target = player.transform; follow.maxX = cameraMax;
            player.GetComponent<PlayerWeapons>().aimCamera = camera;
            var light = new GameObject("Directional Light").AddComponent<Light>(); light.type = LightType.Directional; light.intensity = 1.3f; light.shadows = LightShadows.Soft; light.transform.rotation = Quaternion.Euler(40, -25, 0);
            RenderSettings.ambientLight = new Color(.45f, .5f, .6f);
            var hud = new GameObject("Game HUD").AddComponent<PrototypeHUD>(); hud.motor = player.GetComponent<PlayerMotor>(); hud.weapons = player.GetComponent<PlayerWeapons>(); hud.health = player.GetComponent<CombatHealth>();
            return hud;
        }
        // 이동과 무기를 바로 시험할 수 있는 훈련 씬을 제작합니다.
        public static void BuildTraining()
        {
            Folders();
            var oldScene = SceneManager.GetActiveScene();
            if (string.IsNullOrEmpty(oldScene.path)) EditorSceneManager.SaveScene(oldScene, Root + "/Scenes/OriginalSetup.unity");
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive); SceneManager.SetActiveScene(scene);
            var player = CreatePlayer(); Environment(player, 15);
            var floor = MakeMaterial("Ground", new Color(.18f, .25f, .31f));
            Shape("Ground", PrimitiveType.Cube, new Vector3(5, -.5f, 0), new Vector3(40, 1, 4), floor);
            Shape("Jump platform", PrimitiveType.Cube, new Vector3(7, 1.2f, 0), new Vector3(3, .5f, 3), floor);
            Shape("Left boundary", PrimitiveType.Cube, new Vector3(-15, 4, 0), new Vector3(1, 10, 4), floor);
            Shape("Right boundary", PrimitiveType.Cube, new Vector3(25, 4, 0), new Vector3(1, 10, 4), floor);
            var targetMat = MakeMaterial("Target", new Color(.9f, .3f, .2f));
            foreach (float x in new[] { 3f, 12f, 18f })
            {
                var target = Shape("Practice target", PrimitiveType.Capsule, new Vector3(x, 1, 0), new Vector3(.8f, 1, .8f), targetMat);
                target.layer = 0; var hp = target.AddComponent<CombatHealth>(); hp.practiceTarget = true; hp.maxHealth = 30;
            }
            EditorSceneManager.SaveScene(scene, Root + "/Scenes/MovementWeapons.unity");
            if (oldScene.IsValid() && !oldScene.isDirty) EditorSceneManager.CloseScene(oldScene, true);
            AssetDatabase.SaveAssets();
        }
    }
}
